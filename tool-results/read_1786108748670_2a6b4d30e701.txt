  1067→// ============ PATTERN INSIGHT ============
  1068→function renderInsight() {
  1069→  const container = document.getElementById('insight-container');
  1070→  if (store.urges.length < 2) {
  1071→    container.innerHTML = '';
  1072→    return;
  1073→  }
  1074→  // Find most common words in triggers and emotions
  1075→  const wordCount = {};
  1076→  store.urges.forEach(u => {
  1077→    [u.trigger, u.emotion].forEach(field => {
  1078→      if (!field) return;
  1079→      field.toLowerCase().split(/\s+/).forEach(w => {
  1080→        w = w.replace(/[^a-z]/g, '');
  1081→        if (w.length > 3 && !['that','this','with','have','were','they','them','what','when','from'].includes(w)) {
  1082→          wordCount[w] = (wordCount[w] || 0) + 1;
  1083→        }
  1084→      });
  1085→    });
  1086→  });
  1087→  const sorted = Object.entries(wordCount).sort((a,b) => b[1] - a[1]).slice(0, 3);
  1088→  if (sorted.length === 0 || sorted[0][1] < 2) {
  1089→    container.innerHTML = `
  1090→      <div class="insight-card">
  1091→        <div class="flex items-start gap-3">
  1092→          <i class="fas fa-lightbulb text-[#8aa884] mt-1"></i>
  1093→          <div>
  1094→            <div class="text-xs uppercase tracking-wider text-[#8aa884] mb-1">Pattern Emerging</div>
  1095→            <p class="text-sm text-[#f4ead8]">You've logged ${store.urges.length} urge${store.urges.length>1?'s':''}. Keep journaling — patterns will reveal themselves as you add more entries.</p>
  1096→          </div>
  1097→        </div>
  1098→      </div>`;
  1099→    return;
  1100→  }
  1101→  const top = sorted.map(s => s[0]).join(', ');
  1102→  container.innerHTML = `
  1103→    <div class="insight-card">
  1104→      <div class="flex items-start gap-3">
  1105→        <i class="fas fa-lightbulb text-[#8aa884] mt-1"></i>
  1106→        <div>
  1107→          <div class="text-xs uppercase tracking-wider text-[#8aa884] mb-1">Pattern Detected</div>
  1108→          <p class="text-sm text-[#f4ead8]">Across ${store.urges.length} entries, your most common triggers and emotions involve: <span class="text-[#8aa884] font-semibold">${top}</span>. Awareness is the first step. What will you change?</p>
  1109→        </div>
  1110→      </div>
  1111→    </div>`;
  1112→}
