   626→  const t = audioCtx.currentTime;

   628→  const g = audioCtx.createGain();
   629→  osc.connect(g); g.connect(audioCtx.destination);
   630→  osc.type = 'sine';
   631→  osc.frequency.setValueAtTime(1400, t);
   632→  osc.frequency.exponentialRampToValueAtTime(1800, t + 0.04);
   633→  g.gain.setValueAtTime(0.018, t);
   634→  g.gain.exponentialRampToValueAtTime(0.0001, t + 0.08);
   635→  osc.start(t); osc.stop(t + 0.08);

   660→function sfxOpen() {
   661→  ensureAudio();
   662→  const t = audioCtx.currentTime;
   663→  const len = 0.28;
   664→  const bufSize = Math.floor(audioCtx.sampleRate * len);
   665→  const buf = audioCtx.createBuffer(1, bufSize, audioCtx.sampleRate);
   666→  const d = buf.getChannelData(0);
   667→  for (let i = 0; i < bufSize; i++) {
   668→    d[i] = (Math.random() * 2 - 1) * Math.exp(-i / (audioCtx.sampleRate * 0.04));
   669→  }
   670→  const src = audioCtx.createBufferSource();
   671→  src.buffer = buf;
   672→  const bp = audioCtx.createBiquadFilter();
   673→  bp.type = 'bandpass'; bp.frequency.value = 2800; bp.Q.value = 0.6;
   674→  const g = audioCtx.createGain();
   675→  g.gain.setValueAtTime(0.045, t);
   676→  g.gain.linearRampToValueAtTime(0.06, t + 0.03);
   677→  g.gain.exponentialRampToValueAtTime(0.0001, t + len);
   678→  src.connect(bp); bp.connect(g); g.connect(audioCtx.destination);
   679→  src.start(t);
   680→}

   687→  const buf = audioCtx.createBuffer(1, bufSize, audioCtx.sampleRate);
   688→  const d = buf.getChannelData(0);

   692→  const src = audioCtx.createBufferSource();
   693→  src.buffer = buf;
   694→  const bp = audioCtx.createBiquadFilter();
   695→  bp.type = 'bandpass'; bp.frequency.value = 1400; bp.Q.value = 0.5;
   696→  const g = audioCtx.createGain();
   697→  g.gain.setValueAtTime(0.035, t);

   885→function bindPanelEvents(app) {
   886→  if (app.id === 'notes') {
   887→    setTimeout(() => {
   888→      const area = document.getElementById('noteArea');
   889→      area.addEventListener('input', () => { notesContent = area.value; });
   890→    }, 50);
   891→  }
   892→
   893→  if (app.id === 'calendar') {
   894→    setTimeout(() => {
   895→      const dateIn = document.getElementById('calDate');
   896→      const timeIn = document.getElementById('calTime');
   897→      const eventIn = document.getElementById('calEvent');
   898→      const btn = document.getElementById('calAddBtn');
   899→
   900→      const today = new Date().toISOString().split('T')[0];
   901→      dateIn.value = today;
   902→
   903→      function refreshEvents() {
   904→         const list = document.getElementById('eventList');
   905→         if(calendarEvents.length === 0) {
   906→           list.innerHTML = '<div class="empty-state"><i class="fa-solid fa-calendar-xmark"></i>No events scheduled.</div>';
   907→           return;
   908→         }
   909→         calendarEvents.sort((a,b) => new Date(a.date+'T'+a.time) - new Date(b.date+'T'+b.time));
   910→
   911→         list.innerHTML = calendarEvents.map((ev, i) => `
   912→           <div class="task-item" data-idx="${i}">
   913→             <div style="flex:1;">
   914→               <div class="task-text">${ev.title}</div>
   915→               <div style="font-size:12px; color:var(--text-muted); font-weight: 600;">${ev.date} at ${ev.time}</div>
   916→             </div>
   917→             <button class="task-delete cal-delete" aria-label="Delete event"><i class="fa-solid fa-trash-can"></i></button>
   918→           </div>
   919→         `).join('');
   920→      }
   921→
   922→      btn.addEventListener('click', () => {
   923→        if(!eventIn.value || !dateIn.value || !timeIn.value) {
   924→          showToast('Please fill all fields');
   925→          return;
   926→        }
   927→        calendarEvents.push({
   928→          date: dateIn.value,
   929→          time: timeIn.value,
   930→          title: eventIn.value
   931→        });
   932→        eventIn.value = '';
   933→        sfxClick();
   934→        refreshEvents();
   935→        showToast('Event added');
   936→      });
   937→
   938→      document.getElementById('eventList').addEventListener('click', e => {
   939→        const item = e.target.closest('.task-item');
   940→        if(!item) return;
   941→        if(e.target.closest('.cal-delete')) {
   942→          const idx = parseInt(item.dataset.idx);
   943→          item.style.transform = 'translateX(60px) scale(0.9)';
   944→          item.style.opacity = '0';
   945→          setTimeout(() => {
   946→            calendarEvents.splice(idx, 1);
   947→            sfxClick();
   948→            refreshEvents();
   949→            showToast('Event removed');
   950→          }, 250);
   951→        }
   952→      });
   953→
   954→      refreshEvents();
   955→    }, 50);
   956→  }
   957→}

   985→function updateClock() {
   986→  const now = new Date();
   987→  const h = now.getHours();
   988→  const m = String(now.getMinutes()).padStart(2, '0');
   989→  const s = String(now.getSeconds()).padStart(2, '0');
   990→  const h12 = h % 12 || 12;
   991→  const ampm = h >= 12 ? 'PM' : 'AM';
   992→
   993→  document.getElementById('clockTime').innerHTML = `${h12}:${m}<span class="seconds">${s} ${ampm}</span>`;
   994→  document.getElementById('statusTime').textContent = `${h12}:${m} ${ampm}`;
   995→
   996→  const days = ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'];
   997→  const months = ['January','February','March','April','May','June','July','August','September','October','November','December'];
   998→  document.getElementById('clockDate').textContent = `${days[now.getDay()]}, ${months[now.getMonth()]} ${now.getDate()}`;
   999→
  1000→  let greeting = 'Good Evening';
  1001→  if (h < 12) greeting = 'Good Morning';
  1002→  else if (h < 17) greeting = 'Good Afternoon';
  1003→  document.getElementById('greeting').textContent = greeting;
  1004→}

  1009→function spawnBubble() {
  1010→  const container = document.getElementById('bubbles');
  1011→  const bubble = document.createElement('div');
  1012→  bubble.className = 'bubble';
  1013→  const size = 8 + Math.random() * 28;
  1014→  const left = Math.random() * 100;
  1015→  const dur = 10 + Math.random() * 14;
  1016→  bubble.style.cssText = `
  1017→    width:${size}px; height:${size}px;
  1018→    left:${left}%; bottom:-${size}px;
  1019→    animation-duration:${dur}s;
  1020→  `;
  1021→  container.appendChild(bubble);
  1022→  bubble.addEventListener('animationend', () => {
  1023→    bubble.remove();
  1024→    sfxBubble();
  1025→  });
  1026→}
